cmake_minimum_required(VERSION 3.22)
project(WeaponGrowth VERSION 2.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(wg_core STATIC src/growth/rules.cpp src/catalog/recipe_index.cpp src/storage/ledger.cpp src/gacha/service.cpp)
set_target_properties(wg_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(wg_core PUBLIC src)
if(BUILD_TESTING)
  foreach(name rules runtime gacha command host integration chat_transport)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE wg_core)
    target_include_directories(${name}_test PRIVATE vendor)
    if(name STREQUAL "host" OR name STREQUAL "integration" OR name STREQUAL "chat_transport")
      target_sources(${name}_test PRIVATE src/engine/patch_api.cpp)
    endif()
    if(name STREQUAL "integration")
      target_sources(${name}_test PRIVATE src/runtime/game.cpp src/commands/chat.cpp)
    endif()
    if(name STREQUAL "chat_transport")
      target_sources(${name}_test PRIVATE src/commands/chat.cpp)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
endif()
option(WG_BUILD_MODULE "Build the KernelLoader module" ON)
if(WG_BUILD_MODULE)
  add_library(WeaponGrowth SHARED src/loader/mod.cpp src/engine/patch_api.cpp src/runtime/game.cpp src/commands/chat.cpp)
  target_link_libraries(WeaponGrowth PRIVATE wg_core)
  target_include_directories(WeaponGrowth PRIVATE vendor)
  target_compile_options(WeaponGrowth PRIVATE -O2 -Wall -Wextra)
  target_compile_definitions(WeaponGrowth PRIVATE BUILDING_DLL)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(WG_ARCH x64)
  else()
    set(WG_ARCH x86)
  endif()
  if(ANDROID)
    set(WG_PLATFORM android)
    set(WG_ARCH arm64)
    target_link_options(WeaponGrowth PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
  elseif(WIN32)
    set(WG_PLATFORM windows)
    set_target_properties(WeaponGrowth PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(WG_PLATFORM linux)
  else()
    message(FATAL_ERROR "Supported targets: Android arm64, Windows/Linux x86/x64")
  endif()
  set_target_properties(WeaponGrowth PROPERTIES OUTPUT_NAME "WeaponGrowth.${WG_PLATFORM}.${WG_ARCH}")
endif()
